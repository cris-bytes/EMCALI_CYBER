    epss=number(row.get('epss'),0,1)
    kev_value=row.get('cisa_kev')
    kev_text=str(kev_value).strip().lower() if kev_value is not None else ''
    kev=100. if kev_text in TRUE else 0. if kev_text in FALSE else None
    values={'cvss':cvss*10 if cvss is not None else None,
            'epss':epss*100 if epss is not None else None,
            'kev':kev,
            'asset_criticality':factor_level(row.get('asset_criticality'),LEVELS),
            'exposure':factor_level(row.get('exposure'),EXPOSURE),
            'operational_impact':factor_level(row.get('operational_impact'),LEVELS)}
    missing=[k for k,v in values.items() if v is None]
    weighted=sum(WEIGHTS[k]*v for k,v in values.items() if v is not None)
    known_weight=sum(WEIGHTS[k] for k,v in values.items() if v is not None)
    # Interval bounds explicitly show uncertainty, rather than treating unknown as zero.
    lower=round(weighted,2)
    upper=round(weighted+100*(1-known_weight),2)
    score=round(weighted,2) if not missing else None
    band=('Pendiente de evaluación' if score is None else
          'Crítico' if score>=80 else 'Alto' if score>=60 else 'Medio' if score>=40 else 'Bajo')
    return {'score':score,'band':band,'coverage_pct':round(known_weight*100,1),
            'risk_min':lower,'risk_max':upper,'missing_factors':', '.join(missing),
            'components':values,'method':'V13.3.2 weighted prototype; no empirical calibration'}

from __future__ import annotations
import re

CVE_RE = re.compile(r"^CVE-\d{4}-\d{4,}$", re.I)


def _cve(value):
    if pd.isna(value):
        return ''
    text = str(value).strip().upper()
    return text if CVE_RE.fullmatch(text) else ''


def _numeric(value, low, high):
    try:
        number = float(value)
        return number if pd.notna(number) and low <= number <= high else None
    except (TypeError, ValueError):
        return None


def summarize_provenance(data):
    data = data.copy()
    cvss = pd.to_numeric(data.get('cvss', pd.Series(dtype=float)), errors='coerce')
    epss = pd.to_numeric(data.get('epss', pd.Series(dtype=float)), errors='coerce')
    status = data.get('nvd_status', pd.Series(index=data.index, dtype='object')).astype('string').str.lower()
    source = data.get('cvss_source', pd.Series(index=data.index, dtype='object'))
    date = data.get('epss_date', pd.Series(index=data.index, dtype='object'))
    return {'rows': len(data), 'nvd_analyzed': int(status.eq('analyzed').sum()),
            'cvss_with_source': int((cvss.between(0,10) & source.notna()).sum()),
            'epss_with_date': int((epss.between(0,1) & date.notna()).sum())}

import time
import requests
import pandas as pd
from datetime import datetime, timezone

NVD_URL = "https://services.nvd.nist.gov/rest/json/cves/2.0"
EPSS_URL = "https://api.first.org/data/v1/epss"

def consultar_nvd(cve_id, api_key=None):
    headers = {"User-Agent": "EMCALI-Cyber-Research/13.3.1"}
    if api_key:
        headers["apiKey"] = api_key

    response = requests.get(
        NVD_URL,
        params={"cveId": cve_id},
        headers=headers,
        timeout=30
    )
    response.raise_for_status()

    vulnerabilities = response.json().get("vulnerabilities", [])
    if not vulnerabilities:
        return {"nvd_status": "not_found"}

    record = vulnerabilities[0]["cve"]
    metrics = record.get("metrics", {})

    selected = None
    version = None

    for key, label in [
        ("cvssMetricV40", "4.0"),
        ("cvssMetricV31", "3.1"),
        ("cvssMetricV30", "3.0"),
        ("cvssMetricV2", "2.0"),
    ]:
        candidates = metrics.get(key, [])
        if candidates:
            selected = next(
                (x for x in candidates if x.get("type") == "Primary"),
                candidates[0]
            )
            version = label
            break

    cvss_data = selected.get("cvssData", {}) if selected else {}

    return {
        "nvd_status": record.get("vulnStatus", "unknown"),
        "cvss": cvss_data.get("baseScore"),
        "cvss_version": version,
        "cvss_vector": cvss_data.get("vectorString"),
        "cvss_severity": cvss_data.get("baseSeverity"),
        "cvss_source": selected.get("source") if selected else None,
        "nvd_published": record.get("published"),
        "nvd_modified": record.get("lastModified"),
        "nvd_url": f"https://nvd.nist.gov/vuln/detail/{cve_id}",
    }


def consultar_epss(cve_ids):
    response = requests.get(
        EPSS_URL,
        params={"cve": ",".join(cve_ids)},
        timeout=30
    )
    response.raise_for_status()

    return {
        item["cve"]: {
            "epss": float(item["epss"]),
            "epss_percentile": float(item["percentile"]),
            "epss_date": item.get("date"),
        }
        for item in response.json().get("data", [])
    }


def enriquecer_hallazgos(df, api_key=None):
    result = df.copy()
    if "cve" not in result.columns:
        raise ValueError("Falta la columna cve")
    cves = sorted({
        str(cve).strip().upper()
        for cve in result["cve"].dropna()
        if str(cve).strip().upper().startswith("CVE-")
    })

    epss_data = {}
    epss_error = None
    if cves:
        try:
            epss_data = consultar_epss(cves)
        except requests.RequestException as exc:
            epss_error = str(exc)
    nvd_data = {}

    for cve in cves:
        try:
            nvd_data[cve] = consultar_nvd(cve, api_key)
        except requests.RequestException as exc:
            nvd_data[cve] = {
                "nvd_status": "query_error",
                "nvd_error": str(exc)
            }

        time.sleep(0.6 if api_key else 6.0)

    # Columnas numéricas: CVSS y EPSS
    numeric_columns = [
        "cvss",
        "epss",
        "epss_percentile",
    ]

    # Columnas de texto: versiones, vectores y procedencia
    text_columns = [
        "cvss_version",
        "cvss_vector",
        "cvss_severity",
        "cvss_source",
        "epss_date",
        "nvd_status",
        "nvd_published",
        "nvd_modified",
        "nvd_url",
        "enriched_at",
        "nvd_error",
        "epss_error",
    ]

    for column in numeric_columns:
        if column not in result.columns:
            result[column] = pd.Series(
                float("nan"), index=result.index
            )
        else:
            result[column] = pd.to_numeric(
                result[column], errors="coerce"
            )

    for column in text_columns:
        if column not in result.columns:
            result[column] = pd.Series(
                pd.NA, index=result.index, dtype="object"
            )
        else:
            result[column] = result[column].astype("object")


    for idx, row in result.iterrows():
        cve = str(row.get("cve", "")).strip().upper()
        nvd = nvd_data.get(cve, {})
        epss = epss_data.get(cve, {})

        if nvd.get("nvd_error"):
            result.at[idx, "nvd_error"] = nvd["nvd_error"]
        if epss_error:
            result.at[idx, "epss_error"] = epss_error

        for field in [
            "cvss", "cvss_version", "cvss_vector",
            "cvss_severity", "cvss_source",
            "nvd_status", "nvd_published",
            "nvd_modified", "nvd_url"
        ]:
            if field in nvd and nvd[field] is not None:
                result.at[idx, field] = nvd[field]

        for field, value in epss.items():
            result.at[idx, field] = value

        result.at[idx, "enriched_at"] = (
            datetime.now(timezone.utc).isoformat()
        )

    return result
